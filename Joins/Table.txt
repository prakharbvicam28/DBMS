-- Main tables
CREATE TABLE dept (
    dept_id INT PRIMARY KEY,
    dname VARCHAR(15)
);

CREATE TABLE emp (
    emp_id INT PRIMARY KEY,
    ename VARCHAR(15),
    dept_id INT, -- Foreign key matching dept table
    mgr_id INT   -- For SELF JOIN practice
);
